# Chunk 0_0

Day 0's chunk: the 64 m square at the center of the plain. Coordinates are
local to the chunk's center (+x east, +z south as seen from above, y up);
every built thing is placed by its floor on the Plot at y = 0, and the viewer
sets it down on the ground (lib/terrain.mjs). Nature (the ground, the river,
pines, snow) is not here: it is generated, endlessly, around what is built.

```moxi
from "../lib/cabin.md" import Cabin
from "../lib/well.md" import Well

material Mark { color = "#000000", opacity = 0 }
material Split { color = "#8a5f3a" }
material Bark  { color = "#6b4a2f" }

# day 1: split logs stacked between two stakes, for the cabin's stove
# day 5: the block became real logs, three rows of seven on a bark sleeper
thing Woodpile(rows=3, per=7, r=0.12, step=0.22) {
    part Bed    { shape = box(width=1.8, height=0.08, depth=0.7), material = Bark }
    part StakeW { shape = box(width=0.1, height=1.1, depth=0.1), material = Bark }
    part StakeE { shape = box(width=0.1, height=1.1, depth=0.1), material = Bark }
    anchor floor = Bed.bottom
    relation {
        StakeW.bottom on Bed.top shift=(-0.85, 0) gap=-0.08
        StakeE.bottom on Bed.top shift=(0.85, 0) gap=-0.08
    }
    for k in 0..rows * per {
        let row = floor(k / per)
        let col = k - row * per
        part Log[k] { shape = spin(at(cylinder(height=0.62, radius=r), y=-0.31), axis=x, degrees=90), material = Split }
        relation { Log[k].center on Bed.point(x=(col - (per - 1) / 2) * step, y=0.04 + r - 0.01 + row * (2 * r - 0.03), z=0, nx=0, ny=1, nz=0) }
    }
    resolve grain = 0.05
}

material Plank { color = "#9c7a54" }

# day 6: a log bench by the sauna's door, to sit and cool off facing the river
thing Bench {
    part LegW  { shape = cylinder(height=0.42, radius=0.14), material = Bark }
    part LegE  { shape = cylinder(height=0.42, radius=0.14), material = Bark }
    part Seat  { shape = box(width=1.8, height=0.08, depth=0.45), material = Plank }
    part PostW { shape = box(width=0.08, height=0.6, depth=0.08), material = Bark }
    part PostE { shape = box(width=0.08, height=0.6, depth=0.08), material = Bark }
    part Back  { shape = box(width=1.8, height=0.22, depth=0.06), material = Plank }
    anchor floor = LegW.bottom
    relation {
        Seat.bottom  on LegW.top shift=(0.7, 0)
        LegE.top     on Seat.bottom shift=(0.7, 0)
        PostW.bottom on Seat.top shift=(-0.7, -0.18) gap=-0.02
        PostE.bottom on Seat.top shift=(0.7, -0.18) gap=-0.02
        Back.bottom  on Seat.top shift=(0, -0.18) gap=0.3
    }
    resolve grain = 0.05
}

material Stockfish { color = "#b9ab8f" }

# day 8: a drying rack by the bank, for fish taken through the river's ice
thing FishRack(n=6, span=3.1) {
    part PostW   { shape = box(width=0.12, height=2.2, depth=0.12), material = Bark }
    part RailTop { shape = box(width=span + 0.2, height=0.1, depth=0.1), material = Bark }
    part PostE   { shape = box(width=0.12, height=2.2, depth=0.12), material = Bark }
    part RailLow { shape = box(width=span - 0.1, height=0.08, depth=0.08), material = Bark }
    anchor floor = PostW.bottom
    relation {
        RailTop.bottom on PostW.top shift=(span / 2, 0) gap=-0.05
        PostE.top      on RailTop.bottom shift=(span / 2, 0) gap=-0.05
        RailLow.west   on PostW.east shift=(0.35, 0) gap=-0.02
    }
    for k in 0..n {
        part Fish[k] { shape = ellipsoid(rx=0.08, ry=0.3, rz=0.05), material = Stockfish }
        relation { Fish[k].top on RailTop.bottom shift=((k - (n - 1) / 2) * (span - 0.6) / (n - 1), 0) gap=-0.04 }
    }
    resolve grain = 0.05
}

material Stone { color = "#7c7a74" }
material Sod   { color = "#5d7a3a" }

# day 9: a storehouse up on posts and foot stones, to keep the dried fish and
# the winter's food off the damp ground and out of reach of animals
thing Storehouse(w=3.6, h=2.8, d=3.6, wall=0.2, dw=0.95, dh=2.1) {
    part Deck { shape = box(width=w + 0.2, height=0.2, depth=d + 0.2), material = Plank }
    part Body {
        shape = difference(
            box(width=w, height=h, depth=d),
            box(width=w - 2 * wall, height=h - 0.3, depth=d - 2 * wall),
            at(box(width=dw, height=dh, depth=wall * 3), y=-h / 2 + 0.15 + dh / 2, z=d / 2)
        ),
        material = Bark
    }
    let r = w * 0.92
    part Roof {
        shape = intersect(
            at(box(width=r * 2, height=r, depth=d + 1), y=r / 2),
            scale(spin(box(width=r, height=r, depth=d + 0.8), axis=z, degrees=45), y=0.5)
        ),
        material = Sod
    }
    # the door hung on its hinge (as the cabin's): a point at its bottom hinge
    # corner, set in the doorway, so `twist` turns it about that upright
    part Door    { shape = box(width=dw, height=dh, depth=0.08), material = Plank }
    part StepHi  { shape = box(width=1.2, height=1.2, depth=0.4), material = Stone }
    part StepMid { shape = box(width=1.2, height=0.8, depth=0.4), material = Stone }
    part StepLo  { shape = box(width=1.2, height=0.4, depth=0.4), material = Stone }
    relation {
        Body.bottom    on Deck.top
        Roof.bottom    on Body.top
        Door.point(x=-dw / 2, y=-dh / 2, z=0, nx=0, ny=-1, nz=0)
            on Body.point(x=-dw / 2, y=-h / 2 + 0.15, z=d / 2 - wall / 2, nx=0, ny=1, nz=0)
        StepHi.south   on Deck.north shift=(-0.5, 0)
        StepMid.south  on StepHi.north shift=(-0.2, 0)
        StepLo.south   on StepMid.north shift=(-0.2, 0)
    }
    for k in 0..4 {
        let sx = if k < 2 { -1 } else { 1 }
        let sz = if k - floor(k / 2) * 2 == 0 { -1 } else { 1 }
        part Post[k] { shape = cylinder(height=0.7, radius=0.15), material = Bark }
        part Foot[k] { shape = box(width=0.5, height=0.4, depth=0.5), material = Stone }
        relation {
            Post[k].top on Deck.bottom shift=(sx * (w / 2 - 0.2), sz * (d / 2 - 0.2))
            Foot[k].top on Post[k].bottom
        }
    }
    anchor floor = Deck.point(x=0, y=-1.1, z=0, nx=0, ny=-1, nz=0)

    # the Architect: the door opens (walk into it and it swings; left alone it
    # falls shut), the same hinge and clips as the cabin's
    joint Door { twist=-110..110 }
    pose Opened { Door twist=-95 }
    animate Open  { 0 rest  1.4 Opened }
    animate Close { 0 Opened  1.4 rest }

    resolve grain = 0.05
}

# day 10: a fire ring behind the sauna, to sit by of an evening between the
# hot room and the storehouse: a ring of stones on trodden gravel, logs set
# ready, a log seat on either side
material Gravel { color = "#8d8a80" }
material Ash    { color = "#4a4744" }

thing FireRing(n=9, ring=0.72, logs=4) {
    part Bed { shape = cylinder(height=0.06, radius=1.7), material = Gravel }
    part Hearth { shape = cylinder(height=0.03, radius=0.52), material = Ash }
    part SeatW { shape = spin(at(cylinder(height=1.3, radius=0.19), y=-0.65), axis=x, degrees=90), material = Bark }
    part SeatE { shape = spin(at(cylinder(height=1.3, radius=0.19), y=-0.65), axis=x, degrees=90), material = Bark }
    anchor floor = Bed.bottom
    relation {
        Hearth.bottom on Bed.top
        SeatW.center on Bed.point(x=-1.35, y=0.06 + 0.17, z=0, nx=0, ny=1, nz=0)
        SeatE.center on Bed.point(x=1.35, y=0.06 + 0.17, z=0, nx=0, ny=1, nz=0)
    }
    for k in 0..n {
        part Rock[k] { shape = ellipsoid(rx=0.21, ry=0.15, rz=0.19), material = Stone }
        relation { Rock[k].center on Bed.point(x=ring * cos(360 / n * k), y=0.06 + 0.1, z=ring * sin(360 / n * k), nx=0, ny=1, nz=0) }
    }
    for k in 0..logs {
        part Stick[k] { shape = cylinder(height=0.8, radius=0.05), material = Split }
        relation { Stick[k].bottom on Hearth.top shift=(0.34 * cos(360 / logs * k + 45), 0.34 * sin(360 / logs * k + 45)) gap=-0.02 aim=[0, 0.75, 0] }
    }
    resolve grain = 0.05
}

# day 11: a hay rack on the open ground east of the cabin, the summer's
# grass dried on poles between posts, for the animals the plain will keep
material Hay { color = "#c2ad68" }

thing HayRack(span=4.2, h=1.9) {
    part PostW { shape = cylinder(height=h, radius=0.07), material = Bark }
    part PostA { shape = cylinder(height=h, radius=0.07), material = Bark }
    part PostB { shape = cylinder(height=h, radius=0.07), material = Bark }
    part PostE { shape = cylinder(height=h, radius=0.07), material = Bark }
    part Hay   { shape = box(width=span - 0.2, height=0.85, depth=0.34), material = Hay }
    part Crown { shape = ellipsoid(rx=(span - 0.2) / 2, ry=0.2, rz=0.18), material = Hay }
    relation {
        PostA.center on PostW.point(x=span / 3, y=h / 2, z=0, nx=0, ny=1, nz=0)
        PostB.center on PostW.point(x=2 * span / 3, y=h / 2, z=0, nx=0, ny=1, nz=0)
        PostE.center on PostW.point(x=span, y=h / 2, z=0, nx=0, ny=1, nz=0)
        Hay.center   on PostW.point(x=span / 2, y=1.35, z=0, nx=0, ny=1, nz=0)
        Crown.center on PostW.point(x=span / 2, y=1.76, z=0, nx=0, ny=1, nz=0)
    }
    for k in 0..3 {
        part Pole[k] { shape = spin(at(cylinder(height=span + 0.3, radius=0.04), y=-(span + 0.3) / 2), axis=z, degrees=90), material = Bark }
        relation { Pole[k].center on PostW.point(x=span / 2, y=0.55 + k * 0.5, z=0, nx=0, ny=1, nz=0) }
    }
    anchor floor = PostW.point(x=span / 2, y=0, z=0, nx=0, ny=-1, nz=0)
    resolve grain = 0.05
}

# day 12: a lantern on a post at the path from the cabin's door to the
# well, for fetching water after dark; it is lit at dusk
material Iron  { color = "#3a3a3a" }
material Flame { color = "#e8b04a", emissive = 0 }

thing LanternPost(h=2.6, reach=0.55) {
    part Foot  { shape = box(width=0.5, height=0.9, depth=0.5), material = Stone }
    part Post  { shape = cylinder(height=h, radius=0.08), material = Bark }
    part Arm   { shape = box(width=reach + 0.1, height=0.08, depth=0.08), material = Bark }
    part Hook  { shape = cylinder(height=0.14, radius=0.015), material = Iron }
    part Cap   { shape = cone(height=0.16, radius=0.15), material = Iron }
    part Glass { shape = box(width=0.2, height=0.26, depth=0.2), material = Flame }
    part Base  { shape = box(width=0.24, height=0.04, depth=0.24), material = Iron }
    anchor floor = Post.bottom
    relation {
        Foot.top     on Post.bottom gap=-0.25
        Arm.west     on Post.side(t=0.95, angle=90) gap=-0.08
        Hook.top     on Arm.bottom shift=(reach / 2 - 0.05, 0)
        Cap.apex     on Hook.bottom gap=-0.02
        Glass.top    on Cap.bottom gap=-0.01
        Base.top     on Glass.bottom
    }
    state Lit { Flame.emissive = 2.5 }
    resolve grain = 0.05
}

# day 13: a chopping block beside the woodpile, where the rounds are split
# for the stove and the sauna: a stump on a bed of chips, an axe left in it,
# a round waiting and its two halves fallen to the side
material Chips { color = "#b08a5e" }

thing ChoppingBlock {
    part Bed    { shape = cylinder(height=0.03, radius=0.72), material = Chips }
    part Stump  { shape = cylinder(height=0.5, radius=0.3), material = Bark }
    part Face   { shape = cylinder(height=0.02, radius=0.27), material = Split }
    part Head   { shape = box(width=0.18, height=0.13, depth=0.05), material = Iron }
    part Handle { shape = cylinder(height=0.72, radius=0.03), material = Plank }
    part Round  { shape = cylinder(height=0.42, radius=0.17), material = Bark }
    part HalfN  { shape = intersect(at(box(width=0.34, height=0.17, depth=0.42), y=0.085), spin(at(cylinder(height=0.42, radius=0.17), y=-0.21), axis=x, degrees=90)), material = Split }
    part HalfS  { shape = intersect(at(box(width=0.34, height=0.17, depth=0.42), y=0.085), spin(at(cylinder(height=0.42, radius=0.17), y=-0.21), axis=x, degrees=90)), material = Split }
    anchor floor = Bed.bottom
    relation {
        Stump.bottom  on Bed.top gap=-0.01
        Face.bottom   on Stump.top gap=-0.01
        Head.bottom   on Face.top shift=(0.02, 0) gap=-0.07
        Handle.bottom on Head.east gap=-0.05 lean=(25, 0)
        Round.bottom  on Bed.top shift=(-0.42, 0.38) gap=-0.01
        HalfN.bottom  on Bed.top shift=(-0.38, -0.42) gap=-0.01 twist=20
        HalfS.bottom  on Bed.top shift=(0.1, -0.5) gap=-0.01 twist=-35
    }
    resolve grain = 0.05
}

thing Chunk {
    # the chunk's anchor: everything here hangs off it
    part Plot { shape = box(width=1, height=1, depth=1), material = Mark }

    # day 0: the cabin on the east bank, its door toward the river

    part Wood { thing = Woodpile }

    part Spring { thing = Well }

    part Sauna { thing = Cabin(w=3.8, h=2.9, d=3.4, dw=0.95, dh=2.15) }

    part Rest { thing = Bench }

    part Rack { thing = FishRack }

    part Fire { thing = FireRing }

    # day 13: a chopping block beside the woodpile

    relation {
        Chop.floor on Plot.point(x=20.2, y=0, z=-12.2, nx=0, ny=1, nz=0)

        Fire.floor on Plot.point(x=10, y=0, z=-21, nx=0, ny=1, nz=0)
        Rest.floor on Plot.point(x=5, y=0, z=-14, nx=0, ny=1, nz=0) facing=west
        Wood.floor on Plot.point(x=16, y=0, z=-12, nx=0, ny=1, nz=0)
        Spring.floor on Plot.point(x=16, y=0, z=1, nx=0, ny=1, nz=0) facing=north
        Home.floor on Plot.point(x=16, y=0, z=-6, nx=0, ny=1, nz=0) facing=west
        Sauna.floor on Plot.point(x=10, y=0, z=-14, nx=0, ny=1, nz=0) facing=west
        Rack.floor on Plot.point(x=2.5, y=0, z=-17, nx=0, ny=1, nz=0)
        Store.floor on Plot.point(x=17, y=0, z=-18, nx=0, ny=1, nz=0) facing=north
    }

    resolve grain = 0.5
}
```
