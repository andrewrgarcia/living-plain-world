# Chunk 0_0

Day 0's chunk: the 64 m square at the center of the plain. Coordinates are
local to the chunk's center (+x east, +z south as seen from above, y up);
every built thing is placed by its floor on the Plot at y = 0, and the viewer
sets it down on the ground (lib/terrain.mjs). Nature (the ground, the river,
pines, snow) is not here: it is generated, endlessly, around what is built.

```moxi
from "../lib/cabin.md" import Cabin
from "../lib/well.md" import Well

material Mark { color = "#000000", opacity = 0 }
material Split { color = "#8a5f3a" }
material Bark  { color = "#6b4a2f" }

# day 1: split logs stacked between two stakes, for the cabin's stove
# day 5: the block became real logs, three rows of seven on a bark sleeper
thing Woodpile(rows=3, per=7, r=0.12, step=0.22) {
    part Bed    { shape = box(width=1.8, height=0.08, depth=0.7), material = Bark }
    part StakeW { shape = box(width=0.1, height=1.1, depth=0.1), material = Bark }
    part StakeE { shape = box(width=0.1, height=1.1, depth=0.1), material = Bark }
    anchor floor = Bed.bottom
    relation {
        StakeW.bottom on Bed.top shift=(-0.85, 0) gap=-0.08
        StakeE.bottom on Bed.top shift=(0.85, 0) gap=-0.08
    }
    for k in 0..rows * per {
        let row = floor(k / per)
        let col = k - row * per
        part Log[k] { shape = spin(at(cylinder(height=0.62, radius=r), y=-0.31), axis=x, degrees=90), material = Split }
        relation { Log[k].center on Bed.point(x=(col - (per - 1) / 2) * step, y=0.04 + r - 0.01 + row * (2 * r - 0.03), z=0, nx=0, ny=1, nz=0) }
    }
    resolve grain = 0.05
}

material Plank { color = "#9c7a54" }

# day 6: a log bench by the sauna's door, to sit and cool off facing the river
thing Bench {
    part LegW  { shape = cylinder(height=0.42, radius=0.14), material = Bark }
    part LegE  { shape = cylinder(height=0.42, radius=0.14), material = Bark }
    part Seat  { shape = box(width=1.8, height=0.08, depth=0.45), material = Plank }
    part PostW { shape = box(width=0.08, height=0.6, depth=0.08), material = Bark }
    part PostE { shape = box(width=0.08, height=0.6, depth=0.08), material = Bark }
    part Back  { shape = box(width=1.8, height=0.22, depth=0.06), material = Plank }
    anchor floor = LegW.bottom
    relation {
        Seat.bottom  on LegW.top shift=(0.7, 0)
        LegE.top     on Seat.bottom shift=(0.7, 0)
        PostW.bottom on Seat.top shift=(-0.7, -0.18) gap=-0.02
        PostE.bottom on Seat.top shift=(0.7, -0.18) gap=-0.02
        Back.bottom  on Seat.top shift=(0, -0.18) gap=0.3
    }
    resolve grain = 0.05
}

material Stockfish { color = "#b9ab8f" }

# day 8: a drying rack by the bank, for fish taken through the river's ice
thing FishRack(n=6, span=3.1) {
    part PostW   { shape = box(width=0.12, height=2.2, depth=0.12), material = Bark }
    part RailTop { shape = box(width=span + 0.2, height=0.1, depth=0.1), material = Bark }
    part PostE   { shape = box(width=0.12, height=2.2, depth=0.12), material = Bark }
    part RailLow { shape = box(width=span - 0.1, height=0.08, depth=0.08), material = Bark }
    anchor floor = PostW.bottom
    relation {
        RailTop.bottom on PostW.top shift=(span / 2, 0) gap=-0.05
        PostE.top      on RailTop.bottom shift=(span / 2, 0) gap=-0.05
        RailLow.west   on PostW.east shift=(0.35, 0) gap=-0.02
    }
    for k in 0..n {
        part Fish[k] { shape = ellipsoid(rx=0.08, ry=0.3, rz=0.05), material = Stockfish }
        relation { Fish[k].top on RailTop.bottom shift=((k - (n - 1) / 2) * (span - 0.6) / (n - 1), 0) gap=-0.04 }
    }
    resolve grain = 0.05
}

thing Chunk {
    # the chunk's anchor: everything here hangs off it
    part Plot { shape = box(width=1, height=1, depth=1), material = Mark }

    # day 0: the cabin on the east bank, its door toward the river
    part Home { thing = Cabin }

    part Wood { thing = Woodpile }

    # day 3: a stone well south of the cabin (made a library part on day 7)
    part Spring { thing = Well }

    # day 4: a small sauna toward the river, warmed by the woodpile's logs
    part Sauna { thing = Cabin(w=3.8, h=2.9, d=3.4, dw=0.95, dh=2.15) }

    # day 6: a bench outside the sauna's door, facing the frozen river
    part Rest { thing = Bench }

    # day 8: a rack of drying fish on the bank below the sauna
    part Rack { thing = FishRack }

    relation {
        Rest.floor on Plot.point(x=5, y=0, z=-14, nx=0, ny=1, nz=0) facing=west
        Wood.floor on Plot.point(x=16, y=0, z=-12, nx=0, ny=1, nz=0)
        Spring.floor on Plot.point(x=16, y=0, z=1, nx=0, ny=1, nz=0) facing=north
        Home.floor on Plot.point(x=16, y=0, z=-6, nx=0, ny=1, nz=0) facing=west
        Sauna.floor on Plot.point(x=10, y=0, z=-14, nx=0, ny=1, nz=0) facing=west
        Rack.floor on Plot.point(x=2.5, y=0, z=-17, nx=0, ny=1, nz=0)
    }

    resolve grain = 0.5
}

print Chunk detail=low
```
