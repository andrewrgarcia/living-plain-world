# Chunk 1_0

The chunk east of day 0's: the open pasture beyond the hay rack, where the
plain's first farm begins. Coordinates are local to the chunk's center (+x
east, +z south as seen from above, y up); every built thing is placed by its
floor on the Plot at y = 0, and the viewer sets it down on the ground
(lib/terrain.mjs). Nature is not here: it is generated around what is built.

```moxi
from "../lib/well.md" import Well

material Mark  { color = "#000000", opacity = 0 }
material Log   { color = "#6b4a2f" }
material Sod   { color = "#5d7a3a" }
material Stone { color = "#7c7a74" }
material Plank { color = "#8a5f3a" }
material Dark  { color = "#2b2620" }

# day 14: a byre on the pasture east of the hay rack, the first building of
# a farm, so the animals the hay was dried for have a roof over the winter:
# log walls on a stone plinth, a sod roof, a wide plank door in the gable
# end toward the settlement, a loft hatch above it for pitching in the hay,
# a stone ramp to the door and a small vent window in each long wall
thing Byre(w=5.2, h=3.0, d=7.0, wall=0.22, dw=1.4, dh=2.2) {
    part Body {
        shape = difference(
            box(width=w, height=h, depth=d),
            box(width=w - 2 * wall, height=h - 0.3, depth=d - 2 * wall),
            at(box(width=dw + 0.05, height=dh + 0.02, depth=wall * 3), y=-h / 2 + 0.15 + dh / 2, z=d / 2)
        ),
        material = Log
    }
    let r = w * 0.92
    part Roof {
        shape = intersect(
            at(box(width=r * 2, height=r, depth=d + 1), y=r / 2),
            scale(spin(box(width=r, height=r, depth=d + 0.8), axis=z, degrees=45), y=0.5)
        ),
        material = Sod
    }
    part Plinth { shape = box(width=w + 0.4, height=1.5, depth=d + 0.4), material = Stone }
    part Door   { shape = box(width=dw, height=dh, depth=0.08), material = Plank }
    part Ramp   { shape = box(width=dw + 0.6, height=0.2, depth=0.9), material = Stone }
    part Loft   { shape = box(width=0.9, height=0.7, depth=0.08), material = Plank }
    part VentE  { shape = box(width=0.06, height=0.35, depth=0.7), material = Dark }
    part VentW  { shape = box(width=0.06, height=0.35, depth=0.7), material = Dark }
    relation {
        Roof.bottom   on Body.top
        Plinth.top    on Body.bottom
        Door.point(x=-dw / 2, y=-dh / 2, z=0, nx=0, ny=-1, nz=0)
            on Body.point(x=-dw / 2, y=-h / 2 + 0.15, z=d / 2 - wall / 2, nx=0, ny=1, nz=0)
        Ramp.south    on Body.north shift=(-h / 2 + 0.1, 0) gap=-0.02
        Loft.south    on Roof.north shift=(-r / 2 + 0.55, 0) gap=-0.12
        VentE.west    on Body.east shift=(0.7, 0) gap=-0.03
        VentW.east    on Body.west shift=(0.7, 0) gap=-0.03
    }
    anchor floor = Body.bottom

    # day 16: the byre's door hung on its hinge, so the animals can be let in
    # and out (walk into it and it swings; left alone it falls shut)
    joint Door { twist=-110..110 }
    pose Opened { Door twist=-95 }
    animate Open  { 0 rest  1.4 Opened }
    animate Close { 0 Opened  1.4 rest }
    resolve grain = 0.05
}

# day 15: a paddock beside the byre, so the animals can stand out in the
# open on fair days: split-rail fence, log posts driven 0.3 m into the ground
# and two plank rails between them, a gateway left in the west side toward
# the byre's door
thing Paddock(w=10, d=8, sink=0.3, tall=1.3) {
    let xs = [w / 3, 2 * w / 3, w]
    let zs = [d / 3, 2 * d / 3]
    part Corner { shape = cylinder(height=sink + tall, radius=0.09), material = Log }
    part PostSW { shape = cylinder(height=sink + tall, radius=0.09), material = Log }
    relation { PostSW.bottom on Corner.point(x=0, y=0, z=d, nx=0, ny=1, nz=0) }
    for i in 0..3 {
        part PostN[i] { shape = cylinder(height=sink + tall, radius=0.09), material = Log }
        part PostS[i] { shape = cylinder(height=sink + tall, radius=0.09), material = Log }
        relation {
            PostN[i].bottom on Corner.point(x=xs[i], y=0, z=0, nx=0, ny=1, nz=0)
            PostS[i].bottom on Corner.point(x=xs[i], y=0, z=d, nx=0, ny=1, nz=0)
        }
    }
    for i in 0..2 {
        part PostE[i] { shape = cylinder(height=sink + tall, radius=0.09), material = Log }
        part PostW[i] { shape = cylinder(height=sink + tall, radius=0.09), material = Log }
        relation {
            PostE[i].bottom on Corner.point(x=w, y=0, z=zs[i], nx=0, ny=1, nz=0)
            PostW[i].bottom on Corner.point(x=0, y=0, z=zs[i], nx=0, ny=1, nz=0)
        }
    }
    let ys = [sink + 0.5, sink + 1.05]
    for k in 0..2 {
        part RailN[k]  { shape = box(width=w, height=0.12, depth=0.07), material = Plank }
        part RailS[k]  { shape = box(width=w, height=0.12, depth=0.07), material = Plank }
        part RailE[k]  { shape = box(width=0.07, height=0.12, depth=d), material = Plank }
        part RailWN[k] { shape = box(width=0.07, height=0.12, depth=d / 3), material = Plank }
        part RailWS[k] { shape = box(width=0.07, height=0.12, depth=d / 3), material = Plank }
        relation {
            RailN[k].center  on Corner.point(x=w / 2, y=ys[k], z=0, nx=0, ny=1, nz=0)
            RailS[k].center  on Corner.point(x=w / 2, y=ys[k], z=d, nx=0, ny=1, nz=0)
            RailE[k].center  on Corner.point(x=w, y=ys[k], z=d / 2, nx=0, ny=1, nz=0)
            RailWN[k].center on Corner.point(x=0, y=ys[k], z=d / 6, nx=0, ny=1, nz=0)
            RailWS[k].center on Corner.point(x=0, y=ys[k], z=5 * d / 6, nx=0, ny=1, nz=0)
        }
    }
    anchor floor = Corner.point(x=0, y=sink, z=0, nx=0, ny=-1, nz=0)
    resolve grain = 0.05
}

thing Chunk {
    # the chunk's anchor: everything here hangs off it
    part Plot { shape = box(width=1, height=1, depth=1), material = Mark }

    # day 14: a byre east of the hay rack, its door toward the settlement
    # day 16: its door hung on its hinge
    part Barn { thing = Byre }

    # day 15: a paddock south of the byre, its gateway toward the byre's door
    part Pen { thing = Paddock }

    # day 18: a well east of the paddock, so the farm's animals have water
    # without the walk to the cabin's well
    part FieldWell { thing = Well }

    relation {
        Pen.floor on Plot.point(x=-30, y=0, z=6, nx=0, ny=1, nz=0) facing=north
        Barn.floor on Plot.point(x=-25, y=0, z=0, nx=0, ny=1, nz=0) facing=west
        FieldWell.floor on Plot.point(x=-16, y=0, z=10, nx=0, ny=1, nz=0) facing=west
    }

    resolve grain = 0.5
}

print Chunk detail=low
```
