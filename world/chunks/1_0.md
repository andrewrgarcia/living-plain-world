# Chunk 1_0

The chunk east of day 0's: the open pasture beyond the hay rack, where the
plain's first farm begins. Coordinates are local to the chunk's center (+x
east, +z south as seen from above, y up); every built thing is placed by its
floor on the Plot at y = 0, and the viewer sets it down on the ground
(lib/terrain.mjs). Nature is not here: it is generated around what is built.

```moxi
material Mark  { color = "#000000", opacity = 0 }
material Log   { color = "#6b4a2f" }
material Sod   { color = "#5d7a3a" }
material Stone { color = "#7c7a74" }
material Plank { color = "#8a5f3a" }
material Dark  { color = "#2b2620" }

# day 14: a byre on the pasture east of the hay rack, the first building of
# a farm, so the animals the hay was dried for have a roof over the winter:
# log walls on a stone plinth, a sod roof, a wide plank door in the gable
# end toward the settlement, a loft hatch above it for pitching in the hay,
# a stone ramp to the door and a small vent window in each long wall
thing Byre(w=5.2, h=3.0, d=7.0, wall=0.22, dw=1.4, dh=2.2) {
    part Body {
        shape = difference(
            box(width=w, height=h, depth=d),
            box(width=w - 2 * wall, height=h - 0.3, depth=d - 2 * wall),
            at(box(width=dw + 0.05, height=dh + 0.02, depth=wall * 3), y=-h / 2 + 0.15 + dh / 2, z=d / 2)
        ),
        material = Log
    }
    let r = w * 0.92
    part Roof {
        shape = intersect(
            at(box(width=r * 2, height=r, depth=d + 1), y=r / 2),
            scale(spin(box(width=r, height=r, depth=d + 0.8), axis=z, degrees=45), y=0.5)
        ),
        material = Sod
    }
    part Plinth { shape = box(width=w + 0.4, height=1.5, depth=d + 0.4), material = Stone }
    part Door   { shape = box(width=dw, height=dh, depth=0.08), material = Plank }
    part Ramp   { shape = box(width=dw + 0.6, height=0.2, depth=0.9), material = Stone }
    part Loft   { shape = box(width=0.9, height=0.7, depth=0.08), material = Plank }
    part VentE  { shape = box(width=0.06, height=0.35, depth=0.7), material = Dark }
    part VentW  { shape = box(width=0.06, height=0.35, depth=0.7), material = Dark }
    relation {
        Roof.bottom   on Body.top
        Plinth.top    on Body.bottom
        Door.point(x=-dw / 2, y=-dh / 2, z=0, nx=0, ny=-1, nz=0)
            on Body.point(x=-dw / 2, y=-h / 2 + 0.15, z=d / 2 - wall / 2, nx=0, ny=1, nz=0)
        Ramp.south    on Body.north shift=(-h / 2 + 0.1, 0) gap=-0.02
        Loft.south    on Roof.north shift=(-r / 2 + 0.55, 0) gap=-0.12
        VentE.west    on Body.east shift=(0.7, 0) gap=-0.03
        VentW.east    on Body.west shift=(0.7, 0) gap=-0.03
    }
    anchor floor = Body.bottom
    resolve grain = 0.05
}

thing Chunk {
    # the chunk's anchor: everything here hangs off it
    part Plot { shape = box(width=1, height=1, depth=1), material = Mark }

    # day 14: a byre east of the hay rack, its door toward the settlement
    part Barn { thing = Byre }

    relation {
        Barn.floor on Plot.point(x=-25, y=0, z=0, nx=0, ny=1, nz=0) facing=west
    }

    resolve grain = 0.5
}

print Chunk detail=low
```
